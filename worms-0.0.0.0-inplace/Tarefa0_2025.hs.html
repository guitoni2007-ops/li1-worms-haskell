   14 -- | Retorna a quantidade de munições disponíveis de uma minhoca para uma dada arma.
   15 encontraQuantidadeArmaMinhoca :: TipoArma -> Minhoca -> Int
   16 encontraQuantidadeArmaMinhoca arma minhoca =
   17                     case arma of
   18                         Jetpack -> jetpackMinhoca minhoca
   19                         Escavadora -> escavadoraMinhoca minhoca
   20                         Bazuca -> bazucaMinhoca minhoca
   21                         Mina -> minaMinhoca minhoca
   22                         Dinamite -> dinamiteMinhoca minhoca
   23 -- | Atualiza a quantidade de munições disponíveis de uma minhoca para uma dada arma.
   24 atualizaQuantidadeArmaMinhoca :: TipoArma -> Minhoca -> Int -> Minhoca
   25 atualizaQuantidadeArmaMinhoca arma minhoca novaQuantidade =
   26                  case arma of
   27                     Jetpack -> minhoca { jetpackMinhoca    = novaQuantidade }
   28                     Escavadora -> minhoca { escavadoraMinhoca = novaQuantidade }
   29                     Bazuca     -> minhoca { bazucaMinhoca     = novaQuantidade }
   30                     Mina       -> minhoca { minaMinhoca       = novaQuantidade }
   31                     Dinamite   -> minhoca { dinamiteMinhoca   = novaQuantidade }
   32 -- | Incrementa ou decrementa a quantidade de munições de uma arma numa minhoca.
   33 alteraQuantidadeArmaMinhoca :: TipoArma -> Int -> Minhoca -> Minhoca
   34 alteraQuantidadeArmaMinhoca arma delta minhoca =
   35     let quantidadeAtual = encontraQuantidadeArmaMinhoca arma minhoca
   36         novaQuantidade  = quantidadeAtual + delta
   37     in atualizaQuantidadeArmaMinhoca arma minhoca novaQuantidade
   38 -- | Dispara uma arma, reduzindo automaticamente a munição em 1.
   39 disparaArma :: TipoArma -> Minhoca -> Minhoca
   40 disparaArma arma minhoca =
   41     let qtdAtual = encontraQuantidadeArmaMinhoca arma minhoca
   42         novaQtd = max 0 (qtdAtual - 1)  -- garante que não fica negativa
   43     in atualizaQuantidadeArmaMinhoca arma minhoca novaQtd

   45 -- | Verifica se um tipo de terreno é destrutível, i.e., pode ser destruído por explosões.
   46 --
   47 -- __NB:__ Apenas @Terra@ é destrutível.
   48 eTerrenoDestrutivel :: Terreno -> Bool
   49 eTerrenoDestrutivel Terra = True
   50 eTerrenoDestrutivel _     = False   
   51 
   52 -- | Verifica se um tipo de terreno é opaco, i.e., não permite que objetos ou minhocas se encontrem por cima dele.
   53 --
   54 -- __NB:__ Apenas @Terra@ ou @Pedra@ são opacos.
   55 eTerrenoOpaco :: Terreno -> Bool
   56 eTerrenoOpaco Terra = True
   57 eTerrenoOpaco Pedra = True
   58 eTerrenoOpaco _     = False
   59 
   60 -- | Verifica se uma posição do mapa está livre, i.e., pode ser ocupada por um objeto ou minhoca.
   61 --
   62 -- __NB:__ Uma posição está livre se não contiver um terreno opaco.
   63 ePosicaoMapaLivre :: Posicao -> Mapa -> Bool
   64 ePosicaoMapaLivre (x, y) mapa =
   65     let terreno = mapa !! y !! x  -- acede à posição (x,y) na matriz
   66     in not (eTerrenoOpaco terreno)
   67 
   68 -- | Verifica se uma posição do estado está livre, i.e., pode ser ocupada por um objeto ou minhoca.
   69 --
   70 -- __NB:__ Uma posição está livre se o mapa estiver livre e se não estiver já uma minhoca ou um barril nessa posição.
   71 ePosicaoEstadoLivre :: Posicao -> Estado -> Bool
   72 ePosicaoEstadoLivre pos estado =
   73     let mapaLivre = ePosicaoMapaLivre pos (mapaEstado estado)
   74         minhocaLivre = all (\m -> posicaoMinhoca m /= Just pos) (minhocasEstado estado)
   75         barrilLivre  = all (\o -> case o of
   76                                     Barril {posicaoBarril = p} -> p /= pos
   77                                     _ -> True) (objetosEstado estado)
   78     in mapaLivre && minhocaLivre && barrilLivre

   80 -- | Verifica se numa lista de objetos já existe um disparo feito para uma dada arma por uma dada minhoca.
   81 minhocaTemDisparo :: TipoArma -> NumMinhoca -> [Objeto] -> Bool
   82 minhocaTemDisparo arma numMinhoca objetos =
   83     any (\o -> case o of
   84                  Disparo { tipoDisparo = t, donoDisparo = d } -> t == arma && d == numMinhoca
   85                  _ -> False)
   86         objetos
