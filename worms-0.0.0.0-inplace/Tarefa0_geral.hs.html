   12 -- | Uma matriz é um conjunto de elementos a duas dimensões.
   13 --
   14 -- Em notação matemática, é geralmente representada por:
   15 --
   16 -- <<https://haslab.github.io/Teaching/LI1/2526/img/matriz.png>>
   17 type Matriz a = [[a]]

   19 -- | Uma posição numa matriz é dada como um par (/linha/,/colunha/).
   20 -- As coordenadas são dois números naturais e começam com (0,0) no canto superior esquerdo, com as linhas incrementando para baixo e as colunas incrementando para a direita.
   21 --
   22 -- <<https://haslab.github.io/Teaching/LI1/2526/img/posicaomatriz.png>>
   23 type Posicao = (Int,Int)

   25 -- | A dimensão de uma matrix dada como um par (/número de linhas/,/número de colunhas/).
   26 type Dimensao = (Int,Int)

   28 -- | Uma direção é dada pela rosa dos ventos. Ou seja, os 4 pontos cardeais e os 4 pontos colaterais.

   31 data Direcao = Norte | Nordeste | Este | Sudeste | Sul | Sudoeste | Oeste | Noroeste
   32     deriving (Eq,Ord,Show,Read,Enum)

   36 -- | Verifica se o indice pertence à lista.
   37 eIndiceListaValido :: Int -> [a] -> Bool
   38 eIndiceListaValido x l = x >= 0 && x < length l

   41 -- | Calcula a dimensão de uma matriz.
   42 --ePosicaoMatrizValida
   43 -- __NB:__ Note que não existem matrizes de dimensão /m * 0/ ou /0 * n/, e que qualquer matriz vazia deve ter dimensão /0 * 0/.
   44 dimensaoMatriz :: Matriz a -> Dimensao
   45 dimensaoMatriz [] = (0, 0)
   46 dimensaoMatriz m  = (length m, length (head m))
   47 
   48 -- | Verifica se a posição pertence à matriz.
   49 ePosicaoMatrizValida :: Posicao -> Matriz a -> Bool 
   50 ePosicaoMatrizValida (l,c) m = eIndiceListaValido l m && eIndiceListaValido c (head m)

   53 -- | Move uma posição uma unidade no sentido de uma direção.
   54 movePosicao :: Direcao -> Posicao -> Posicao
   55 movePosicao d (l,c)   | d==Norte = (l-1,c)
   56                       | d==Sul = (l+1,c)
   57                       | d==Oeste = (l,c-1)
   58                       | d==Este = (l,c+1)
   59                       | d==Nordeste = (l-1,c+1)
   60                       | d==Noroeste = (l-1,c-1)
   61                       | d==Sudeste = (l+1,c+1)
   62                       | d==Sudoeste = (l+1,c-1)
   63              
   64 
   65 -- | Versão da função 'movePosicao' que garante que o movimento não se desloca para fora de uma janela.
   66 --
   67 -- __NB:__ Considere uma janela retangular com origem no canto superior esquerdo definida como uma matriz. A função recebe a dimensao da janela.
   68 movePosicaoJanela :: Dimensao -> Direcao -> Posicao -> Posicao
   69 movePosicaoJanela (x,y) d (l,c) | d==Norte && l > 0 = (l-1,c)
   70                                 | d==Sul && l < x-1 = (l+1,c)
   71                                 | d==Oeste && c > 0 = (l,c-1)
   72                                 | d==Este && c < y-1 = (l,c+1)
   73                                 | d==Nordeste && l > 0 && c < y-1 = (l-1,c+1)
   74                                 | d==Noroeste && l > 0 && c > 0 = (l-1,c-1)
   75                                 | d==Sudeste && l < x-1 && c < y-1 = (l+1,c+1)
   76                                 | d==Sudoeste && l < x-1 &&  c > 0 = (l+1,c-1)
   77

   79 -- | Converte uma posição no referencial em que a origem é no canto superior esquerdo da janela numa posição em que a origem passa a estar no centro da janela.
   80 --
   81 -- __NB:__ Considere posições válidas. Efetue arredondamentos como achar necessário.
   82 origemAoCentro :: Dimensao -> Posicao -> Posicao
   83 origemAoCentro (x,y) (l,c) =
   84   let centroL = x `div` 2
   85       centroC = y `div` 2
   86   in (c - centroC, centroL - l)

   90 -- | Roda um par (posição,direção) 45% para a direita.
   91 --
   92 -- __NB:__ Vendo um par (posição,direção) como um vector, cria um novo vetor do desto com a próxima direção da rosa dos ventos rodando para a direita.
   93 --
   94 -- <<https://haslab.github.io/Teaching/LI1/2526/img/rodaposicaodirecao.png>>
   95 rodaPosicaoDirecao :: (Posicao,Direcao) -> (Posicao,Direcao)
   96 rodaPosicaoDirecao ((l,c),d) | d==Norte = ((l,c), Nordeste)
   97                              | d==Nordeste = ((l,c), Este)
   98                              | d==Este = ((l,c), Sudeste)
   99                              | d==Sudeste = ((l,c), Sul)
  100                              | d==Sul = ((l,c), Sudoeste)
  101                              | d==Sudoeste = ((l,c), Oeste)
  102                              | d==Oeste = ((l,c), Noroeste)
  103                              | d==Noroeste = ((l,c), Norte)

  108 -- | Devolve o elemento num dado índice de uma lista.
  109 --
  110 -- __NB:__ Retorna @Nothing@ se o índice não existir.
  111 encontraIndiceLista :: Int -> [a] -> Maybe a
  112 encontraIndiceLista i l
  113   | i < 0 || i >= length l = Nothing
  114   | otherwise = Just (l !! i)
  115 
  116 
  117 -- | Modifica um elemento num dado índice.
  118 --
  119 -- __NB:__ Devolve a própria lista se o índice não existir.
  120 atualizaIndiceLista :: Int -> a -> [a] -> [a]
  121 atualizaIndiceLista _ _ [] = []
  122 atualizaIndiceLista i x l@(h:t)
  123   | i < 0     = l                    
  124   | i == 0    = x : t                 
  125   | otherwise = h : atualizaIndiceLista (i - 1) x t  
  126

  128 -- | Devolve o elemento numa dada posição de uma matriz.
  129 --
  130 -- __NB:__ Retorna @Nothing@ se a posição não existir.
  131 encontraPosicaoMatriz :: Posicao -> Matriz a -> Maybe a
  132 encontraPosicaoMatriz (l, c) m
  133   | l < 0 || l >= length m = Nothing
  134   | c < 0 || c >= length (m !! l) = Nothing
  135   | otherwise = Just ((m !! l) !! c)
  136 
  137 
  138 -- | Modifica um elemento numa dada posição de uma matriz.
  139 --
  140 -- __NB:__ Devolve a própria matriz se o elemento não existir.
  141 atualizaPosicaoMatriz :: Posicao -> a -> Matriz a -> Matriz a
  142 atualizaPosicaoMatriz _ _ [] = []
  143 atualizaPosicaoMatriz (0, c) e (linha:t) = atualizaIndiceLista c e linha : t
  144 atualizaPosicaoMatriz (l, c) e (linha:t) = linha : atualizaPosicaoMatriz (l - 1, c) e t

  147 -- | Aplica uma sequência de movimentações a uma posição, pela ordem em que ocorrem na lista.
  148 moveDirecoesPosicao :: [Direcao] -> Posicao -> Posicao
  149 moveDirecoesPosicao [] pos = pos                             
  150 moveDirecoesPosicao (h:t) pos = moveDirecoesPosicao t (movePosicao h pos)
  151

  153 -- | Aplica a mesma movimentação a uma lista de posições.
  154 moveDirecaoPosicoes :: Direcao -> [Posicao] -> [Posicao]
  155 moveDirecaoPosicoes _ [] = []                      
  156 moveDirecaoPosicoes d (h:t) = movePosicao d h : moveDirecaoPosicoes d t

  159 -- | Verifica se uma matriz é válida, no sentido em que modela um rectângulo.
  160 -- __NB:__ Todas as linhas devem ter o mesmo número de colunas.
  161 eMatrizValida :: Matriz a -> Bool
  162 eMatrizValida [] = False
  163 eMatrizValida [[]] = False
  164 eMatrizValida [_] = True
  165 eMatrizValida (l1:l2:ls)
  166   | length l1 == length l2 = eMatrizValida (l2:ls)
  167   | otherwise = False
