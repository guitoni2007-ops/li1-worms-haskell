   12 -- | Função principal da Tarefa 1. Recebe um estado e retorna se este é válido ou não.
   13 validaEstado :: Estado -> Bool
   14 validaEstado estado =
   15   let mapa     = mapaEstado estado
   16       objetos  = objetosEstado estado
   17       minhocas = minhocasEstado estado
   18   in validaMapa mapa && validaObj (1,1) mapa objetos minhocas && validaWorms (0,0) mapa objetos minhocas

   21 -- | 1 Função secundária da Tarefa 1. Recebe um mapa e retorna se este é válido ou não.
   22 validaMapa :: Mapa -> Bool
   23 validaMapa mapa = eMatrizValida(mapa) && eTerrenosValido(mapa)

   27 eTerrenosValido :: Mapa -> Bool
   28 eTerrenosValido [] = True
   29 eTerrenosValido (t1:ts) = eTerrenoValido t1 && eTerrenosValido ts

   32 eTerrenoValido :: [Terreno] -> Bool
   33 eTerrenoValido [] = True
   34 eTerrenoValido (t1:ts) = if (t1 == Ar || t1 == Agua || t1 == Terra || t1 == Pedra) then eTerrenoValido(ts)
   35                          else False

   40 -- | 2 Função secundária da Tarefa 1. Verifica se uma lista de objetos é válida ou não.
   41 validaObj :: Posicao -> Mapa -> [Objeto] -> [Minhoca] -> Bool
   42 validaObj posicao mapa objetos minhocas = objetosValidosComExcecaoBazuca mapa objetos && objetosPosicao objetos minhocas && objeto_disparo objetos minhocas
   43 
   44 --Verifica se todas as posicoes dos objetos são válidas, incluindo a exceção da bazuca.
   45 objetosValidosComExcecaoBazuca :: Mapa -> [Objeto] -> Bool
   46 objetosValidosComExcecaoBazuca mapa =
   47   all (objetoValidoComExcecao mapa)

   50 --Verifica se uma posição é válida e livre no mapa.
   51 posicaovalidalivre :: Posicao -> Mapa -> Bool
   52 posicaovalidalivre (l,c) mapa = ePosicaoMatrizValida (l,c) mapa && not (eTerrenoOpaco (mapa !! l !! c))

   54 --Devolve a direção oposta à dada.
   55 direcaoContraria :: Direcao -> Direcao
   56 direcaoContraria Norte     = Sul
   57 direcaoContraria Sul       = Norte
   58 direcaoContraria Este      = Oeste
   59 direcaoContraria Oeste     = Este
   60 direcaoContraria Nordeste  = Sudoeste
   61 direcaoContraria Sudoeste  = Nordeste
   62 direcaoContraria Noroeste  = Sudeste
   63 direcaoContraria Sudeste   = Noroeste

   65 --Devolve o terreno numa posição da matriz, ou Ar se a posição for inválida.
   66 terrenoEm :: Posicao -> Matriz Terreno -> Terreno
   67 terrenoEm pos mapa =
   68   case encontraPosicaoMatriz pos mapa of
   69     Just t  -> t
   70     Nothing -> Ar
   71 
   72 --Permite disparo de bazuca sobre terreno opaco se a posição anterior não for opaca.
   73 bazucaPodeEstarSobreOpaco :: Mapa -> Posicao -> Direcao -> Bool
   74 bazucaPodeEstarSobreOpaco mapa pos dir =
   75   eTerrenoOpaco (terrenoEm pos mapa) &&
   76   not (eTerrenoOpaco (terrenoEm (movePosicao (direcaoContraria dir) pos) mapa))
   77  
   78 --Verifica se uma posição é válida e livre no mapa,
   79 -- ou se pode conter um disparo de bazuca sobre terreno opaco.
   80 posicaoValidaOuBazuca :: Mapa -> Posicao -> Direcao -> Bool
   81 posicaoValidaOuBazuca mapa pos dir =
   82   posicaovalidalivre pos mapa || bazucaPodeEstarSobreOpaco mapa pos dir
   83 
   84 --Verifica se um objeto é válido, considerando a exceção da bazuca.
   85 objetoValidoComExcecao :: Mapa -> Objeto -> Bool
   86 objetoValidoComExcecao mapa (Disparo pos dir Bazuca _ _) =
   87   posicaoValidaOuBazuca mapa pos dir
   88 objetoValidoComExcecao mapa (Disparo pos _ _ _ _) =
   89   posicaovalidalivre pos mapa
   90 objetoValidoComExcecao _ _ = True

   93 --Verifica se duas posições sao diferentes.
   94 posicoesDiferentes :: Posicao -> Posicao -> Bool
   95 posicoesDiferentes p1 p2 = p1 /= p2
   96 
   97 --Verifica se duas posições sao diferentes para maybe posição.
   98 posicoesDiferentes2 :: Posicao -> Maybe Posicao -> Bool
   99 posicoesDiferentes2 p1 (Just p2) = p1 /= p2
  100 posicoesDiferentes2 _ Nothing    = True

  103 --Verificar a posição do barril em relação a todas as minhocas.
  104 verificaPosBarrilMinhoca :: Posicao -> [Minhoca] -> Bool
  105 verificaPosBarrilMinhoca pos1 [] = True

  108 --Verifica se as posições do barris e das minhocas não são iguais.
  109 verificaPosBarril :: Objeto -> [Objeto] -> [Minhoca] -> Bool
  110 verificaPosBarril (Barril pos1 explode1) [] m = verificaPosBarrilMinhoca pos1 m
  111 verificaPosBarril (Barril pos1 explode1) ((Barril pos2 explode2):os) minhocas = posicoesDiferentes pos1 pos2 && verificaPosBarrilMinhoca pos1 minhocas && verificaPosBarril (Barril pos1 explode1) os minhocas

  115 verificaPosMinaDina o [] = True
  116 verificaPosMinaDina (Disparo pos1 dir1 tipo1 tempo1 dono1) ((Barril pos2 explode2):os) = posicoesDiferentes pos1 pos2 && verificaPosMinaDina (Disparo pos1 dir1 tipo1 tempo1 dono1) os

  118 --Filtra uma lista de objetos retornando uma lista só de barris.
  119 filtraBarris :: [Objeto] -> [Objeto]
  120 filtraBarris = filter isBarril
  121   where
  122     isBarril (Barril _ _) = True
  123     isBarril _            = False

  126 --Verifica a posição de objetos especificos (Barril, Dinamite ou Mina).
  127 objetoPosicao :: Objeto -> [Objeto] -> [Minhoca] -> Bool
  128 objetoPosicao objeto listaObjetos minhocas = case objeto of
  129                                         Barril pos explode -> verificaPosBarril objeto (filtraBarris listaObjetos) minhocas
  130 
  131                                         Disparo pos dir tipo tempo dono ->
  132                                             case tipo of
  133                                                 Dinamite -> verificaPosMinaDina objeto (filtraBarris listaObjetos)
  134                                                 Mina     -> verificaPosMinaDina objeto (filtraBarris listaObjetos)
  135                                                 _        -> True

  140 proximoObjeto :: [Objeto] -> Maybe (Objeto, [Objeto])
  141 proximoObjeto []     = Nothing
  142 proximoObjeto (o:os) = Just (o, os)

  145 removerObjeto :: Objeto -> [Objeto] -> [Objeto]
  146 removerObjeto obj = filter (/= obj)

  149 --Recebe uma lista de objetos e passa a "objetoPosicao" um objeto de cada vez.
  150 objetosPosicao :: [Objeto] -> [Minhoca] -> Bool
  151 objetosPosicao todos minhocas = verificar todos
  152   where
  153     verificar [] = True
  154     verificar lista =
  155       case proximoObjeto lista of
  156         Nothing -> True
  157         Just (obj, _) -> objetoPosicao obj (removerObjeto obj todos) minhocas && verificar (tail lista)

  163 --Dado uma lista de disparos, verifica se algum elemento é do tipo de arma: Jetpack ou Escavadora.
  164 dispJetEsc :: [Objeto] -> Bool
  165 dispJetEsc [] = True
  166 dispJetEsc ((Disparo _ _ tipo _ _):os) =
  167   tipo /= Jetpack && tipo /= Escavadora && dispJetEsc os
  168 dispJetEsc (_:os) = dispJetEsc os

  173 --Dado uma lista de disparos, verifica se o tempo do disparo é válido para o tipo de arma.
  174 validatempoDisparo :: [Objeto] -> Bool
  175 validatempoDisparo [] = True
  176 validatempoDisparo ((Disparo _ _ tipo tempo _):os) =
  177   case tempo of
  178     Just t ->
  179       case tipo of
  180         Mina     -> (t >= 0 && t <= 2) && validatempoDisparo os
  181         Dinamite -> (t >= 0 && t <= 4) && validatempoDisparo os
  182         
  183     Nothing ->   
  184       case tipo of
  185         Bazuca -> validatempoDisparo os
  186         Mina -> validatempoDisparo os
  187         
  188 validatempoDisparo (_:os) = validatempoDisparo os
  189 
  190 --Verifica se o dono do disparo tem um índice válido na lista de minhocas.
  191 validaDonoDisparo :: [Objeto] -> [Minhoca] -> Bool
  192 validaDonoDisparo [] _ = True
  193 validaDonoDisparo ((Disparo _ _ _ _ dono):os) minhocas =
  194   eIndiceListaValido dono minhocas && validaDonoDisparo os minhocas
  195 validaDonoDisparo (_:os) minhocas = validaDonoDisparo os minhocas
  196 
  197 --Verifica se o mesmo dono nao tem simultaneamente mais do que um disparo de cada tipo.
  198 maisQueUmDisparo :: [Objeto] -> Bool
  199 maisQueUmDisparo objetos = verifica [] objetos
  200  
  201  --Função auxiiar de maisQueUmDisparo.
  202 verifica :: [(NumMinhoca, TipoArma)] -> [Objeto] -> Bool
  203 verifica _ [] = True
  204 verifica vistos (Disparo _ _ tipo _ dono : os)
  205     | (dono, tipo) `elem` vistos = False
  206     | otherwise = verifica ((dono, tipo) : vistos) os
  207 verifica vistos (_ : os) = verifica vistos os

  213 objeto_disparo objetos minhocas = dispJetEsc objetos && validatempoDisparo objetos && validaDonoDisparo objetos minhocas && maisQueUmDisparo objetos

  215 -- | 3 Função secundária da Tarefa 1. Recebe uma lista de minhocas e retorna se este é válido ou não.
  216 validaWorms :: Posicao -> Mapa -> [Objeto] -> [Minhoca] -> Bool
  217 validaWorms posicao mapa objetos minhocas =
  218   posicaovalidaminhoca minhocas mapa &&
  219   minhocasValidas minhocas objetos mapa &&
  220   minhocamorta mapa minhocas &&
  221   verificavidaminhocas minhocas &&
  222   validaMinhocasMunicao minhocas

  225 --Valida se uma lista de minhocas tem uma posição válida e livre no mapa ou nenhuma posição
  226 posicaovalidaminhoca :: [Minhoca] -> Mapa -> Bool
  227 posicaovalidaminhoca [] _ = True
  228 posicaovalidaminhoca (m:ms) mapa =
  229   case posicaoMinhoca m of
  230     Just pos -> posicaovalidalivre pos mapa && posicaovalidaminhoca ms mapa
  231     Nothing  -> estaMorta m && posicaovalidaminhoca ms mapa
  232

  234 --Verifica se a posição da minhoca não coincide com outras minhocas
  235 verificaPosMinhocaMinhocas :: Minhoca -> [Minhoca] -> Bool
  236 verificaPosMinhocaMinhocas minhoca outras =
  237   case posicaoMinhoca minhoca of
  238     Nothing   -> True
  239     Just pos1 -> all (\m -> posicoesDiferentes2 pos1 (posicaoMinhoca m)) outras
  240 
  241 --Verifica se a posição da minhoca não coincide com nenhum barril
  242 verificaPosMinhocaBarris :: Minhoca -> [Objeto] -> Bool
  243 verificaPosMinhocaBarris minhoca objetos =
  244   case posicaoMinhoca minhoca of
  245     Nothing   -> True
  246     Just pos1 -> all (posicoesDiferentes pos1) (posicoesDosBarris objetos)
  247 
  248 --Função auxiliar a verificaPosMinhocaBarris
  249 posicoesDosBarris :: [Objeto] -> [Posicao]
  250 posicoesDosBarris [] = []
  251 posicoesDosBarris ((Barril p _):os) = p : posicoesDosBarris os
  252 posicoesDosBarris (_:os) = posicoesDosBarris os
  253 
  254 --Junta todas as verificações para uma minhoca
  255 minhocaValida :: Minhoca -> [Minhoca] -> [Objeto] -> Mapa -> Bool
  256 minhocaValida m outras objetos mapa =
  257   verificaPosMinhocaBarris m objetos &&
  258   verificaPosMinhocaMinhocas m outras &&
  259   case posicaoMinhoca m of
  260     Nothing  -> True
  261     Just pos -> posicaovalidalivre pos mapa
  262 
  263 --Aplica a verificação a todas as minhocas
  264 minhocasValidas :: [Minhoca] -> [Objeto] -> Mapa -> Bool
  265 minhocasValidas [] _ _ = True
  266 minhocasValidas (m:ms) objetos mapa =
  267   minhocaValida m ms objetos mapa && minhocasValidas ms objetos mapa
  268 
  269 --Verifica se a vida da minhoca está como Morta
  270 estaMorta :: Minhoca -> Bool
  271 estaMorta minhoca = vidaMinhoca minhoca == Morta
  272 
  273 --Verifica se uma minhoca está morta se estiver fora do mapa ou em água
  274 minhocaEstaCorretamenteMorta :: Mapa -> Minhoca -> Bool
  275 minhocaEstaCorretamenteMorta mapa minhoca =
  276   case posicaoMinhoca minhoca of
  277     Nothing -> estaMorta minhoca
  278     Just pos ->
  279       case encontraPosicaoMatriz pos mapa of
  280         Just Agua -> estaMorta minhoca
  281         _         -> True
  282 
  283 --Verifica se todas as minhocas estão corretamente mortas quando fora do mapa ou em água
  284 minhocamorta :: Mapa -> [Minhoca] -> Bool
  285 minhocamorta mapa = all (minhocaEstaCorretamenteMorta mapa)
  286 
  287 --Verifica se a vida de uma minhoca é válida
  288 verificavidaminhoca :: VidaMinhoca -> Bool
  289 verificavidaminhoca (Viva n) = n >= 0 && n <= 100
  290 verificavidaminhoca Morta    = True

  293 --Verifica se a vida de uma lista de minhocas é válida
  294 verificavidaminhocas :: [Minhoca] -> Bool
  295 verificavidaminhocas [] = True
  296 verificavidaminhocas (m:ms) =
  297   verificavidaminhoca (vidaMinhoca m) && verificavidaminhocas ms

  303   escavadoraMinhoca m >= 0 &&

  311 validaMinhocasMunicao [] = True
